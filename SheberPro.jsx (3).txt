              {/* 1. PHOTOS — сверху */}
              <div className="jd-section" style={{padding:"14px 14px 10px"}}>
                <div className="jd-section-ttl" style={{marginBottom:10}}>{t.photos}</div>
                <div className="photos-row">
                  {openJob.photos.map((p,i)=>(
                    <div key={i} className="photo-box" style={{width:110,height:95}}>{p}</div>
                  ))}
                  <div className="photo-box" style={{width:110,height:95,border:"2px dashed var(--bord)",fontSize:22,color:"var(--muted)"}}>+</div>
                </div>
              </div>

              {/* 2. BUDGET strip */}
              <div style={{background:"var(--ink2)",borderRadius:16,padding:"14px 16px",marginBottom:12,display:"flex",alignItems:"center",justifyContent:"space-between"}}>
                <div>
                  <div style={{fontSize:11,color:"rgba(255,255,255,.5)",marginBottom:3}}>{t.budget}</div>
                  <div style={{fontFamily:"'Unbounded',sans-serif",fontSize:22,fontWeight:900,color:"#fff"}}>{openJob.budget}</div>
                </div>
                {openJob.urgent && <div className="urg-badge" style={{background:"var(--acc)",color:"#fff"}}>{t.urgent}</div>}
              </div>

              {/* 3. DESCRIPTION */}
              <div className="jd-section">
                <div className="jd-section-ttl">{t.description}</div>
                <div className="jd-desc">{openJob.desc[lang]}</div>
              </div>

              {/* 4. ADDRESS + META */}
              <div className="jd-section">
                <div className="jd-row">
                  <div className="jd-row-icon" style={{background:"#F0FDF4"}}>📍</div>
                  <div className="jd-row-text"><h4>{t.district}</h4><p>{openJob.district}, {city}</p></div>
                </div>
                <div className="jd-row">
                  <div className="jd-row-icon" style={{background:"#FFF7ED"}}>💼</div>
                  <div className="jd-row-text"><h4>{t.jobType}</h4><p>{openJob.jobType[lang]}</p></div>
                </div>
                <div className="jd-row">
                  <div className="jd-row-icon" style={{background:"#FFF1F2"}}>🕐</div>
                  <div className="jd-row-text"><h4>{lang==="ru"?"Размещено":lang==="kz"?"Орналастырылды":"Posted"}</h4><p>{openJob.time}</p></div>
                </div>
              </div>

              {/* 5. CLIENT INFO + PHONE */}
              <div className="jd-section">
                <div className="jd-section-ttl">{lang==="ru"?"Заказчик":lang==="kz"?"Тапсырыс беруші":"Client"}</div>
                <div className="client-row" style={{marginBottom:14}}>
                  <div className="cli-avatar">{openJob.client.name[0]}</div>
                  <div className="cli-info">
                    <h4>{openJob.client.name}</h4>
                    <p>{openJob.client.reviews} {lang==="ru"?"отзывов":lang==="kz"?"пікір":"reviews"}</p>
                  </div>
                  <div className="cli-rating">
                    <div className="stars">⭐ {openJob.client.rating}</div>
                  </div>
                </div>
                <button className="phone-btn">
                  <Ic path={icons.phone} size={18} color="var(--green)"/>
                  {openJob.phone}
                </button>
              </div>

              {/* 6. REVIEWS + QUICK CONTACT */}
              <div className="jd-section">
                <div className="jd-section-ttl">{t.clientReviews}</div>
                {openJob.clientReviews.length===0
                  ? <div style={{fontSize:13,color:"var(--muted)",marginBottom:16}}>{t.noReviews}</div>
                  : openJob.clientReviews.map((rv,i)=>(
                    <div key={i} className="rev-item">
                      <div className="rev-top">
                        <div className="rev-name">{rv.name}</div>
                        <div className="rev-stars"><Stars n={rv.stars}/></div>
                      </div>
                      <div className="rev-text">{rv.text[lang]}</div>
                    </div>
                  ))
                }
                {/* QUICK CONTACT */}
                <div style={{marginTop:16,paddingTop:14,borderTop:"1px solid var(--bord)"}}>
                  <div style={{fontSize:11,fontWeight:700,color:"var(--muted)",textTransform:"uppercase",letterSpacing:".8px",marginBottom:10}}>
                    {lang==="ru"?"Быстрая связь":lang==="kz"?"Жылдам байланыс":"Quick contact"}
                  </div>
                  <div style={{display:"flex",gap:10}}>
                    <button onClick={()=>setOpenJob(null)} style={{flex:1,background:"var(--ink2)",color:"#fff",border:"none",borderRadius:14,padding:"14px 10px",fontFamily:"'Unbounded',sans-serif",fontSize:12,fontWeight:700,cursor:"pointer",display:"flex",alignItems:"center",justifyContent:"center",gap:6}}>
                      <Ic path={icons.chat} size={16} color="#fff"/>
                      {lang==="ru"?"Написать":lang==="kz"?"Жазу":"Chat"}
                    </button>
                    <button style={{flex:1,background:"rgba(16,185,129,.1)",color:"var(--green)",border:"1.5px solid rgba(16,185,129,.25)",borderRadius:14,padding:"14px 10px",fontFamily:"'Unbounded',sans-serif",fontSize:12,fontWeight:700,cursor:"pointer",display:"flex",alignItems:"center",justifyContent:"center",gap:6}}>
                      <Ic path={icons.phone} size={16} color="var(--green)"/>
                      {lang==="ru"?"Позвонить":lang==="kz"?"Қоңырау шалу":"Call"}
                    </button>
                    <button style={{flex:1,background:"rgba(37,211,102,.1)",color:"#25D366",border:"1.5px solid rgba(37,211,102,.25)",borderRadius:14,padding:"14px 10px",fontFamily:"'Unbounded',sans-serif",fontSize:12,fontWeight:700,cursor:"pointer",display:"flex",alignItems:"center",justifyContent:"center",gap:6}}>
                      <span style={{fontSize:16}}>💬</span>
                      WhatsApp
                    </button>
                  </div>
                </div>
              </div>

            </div>
            <div className="jd-respond-btn">
              <button className="jd-respond-inner" onClick={()=>setOpenJob(null)}>{t.respond} →</button>
            </div>
          </div>
        )}

        {/* ── TARIFF CHANGE MODAL ── */}
        {showTariffChange && (
          <div className="trf-modal fi">
            <div className="tm-hdr">
              <button className="back-btn" onClick={()=>setShowTariffChange(false)}><Ic path={icons.x} size={18}/></button>
              <div style={{fontFamily:"'Unbounded',sans-serif",fontSize:17,fontWeight:700,color:"#fff"}}>{t.changeTariff}</div>
            </div>
            <div className="trf-body">
              {TARIFFS.map(tr=>(
                <div key={tr.id} className={`trf-card ${selTariff===tr.id?"sel":""} ${tr.id==="months3"?"hot":""}`} onClick={()=>setSelTariff(tr.id)}>
                  {tr.gift && <div className="trf-gift-badge">🎁 {tr.gift[lang]}</div>}
                  {!tr.gift && tr.badge && <div className="trf-badge">{tr.badge}</div>}
                  <div className="trf-row">
                    <div className="trf-name">{lang==="ru"?tr.ru:lang==="kz"?tr.kz:tr.en}</div>
                    <div className="trf-price">{tr.price} <span>тг</span></div>
                  </div>
                  {tr.gift && <div className="trf-gift">🎁 {tr.gift[lang]}</div>}
                  <div className="trf-days">{tr.sub ? tr.sub[lang] : tr.days+" "+t.totalDays}</div>
                </div>
              ))}
              <button className="act-btn" onClick={()=>setShowTariffChange(false)}>{t.activate} →</button>
            </div>
          </div>
        )}

        {/* ── POST JOB MODAL ── */}
        {showPost && (
          <div className="post-modal fi">
            <div className="pm-hdr">
              <button className="back-btn" onClick={()=>setShowPost(false)}><Ic path={icons.x} size={18}/></button>
              <div className="pm-ttl">{t.postJob}</div>
            </div>
            <div className="pm-body">
              <div className="f-grp">
                <label className="f-lbl">{t.jobType}</label>
                <div className="jtype-btns">
                  <div className={`jtype-btn ${jobType==="oneTime"?"sel":""}`} onClick={()=>setJobType("oneTime")}>
                    <span className="jtype-icon">🔨</span><span className="jtype-name">{t.oneTime}</span>
                  </div>
                  <div className={`jtype-btn ${jobType==="salary"?"sel":""}`} onClick={()=>setJobType("salary")}>
                    <span className="jtype-icon">💼</span><span className="jtype-name">{t.salary}</span>
                  </div>
                </div>
              </div>
              <div className="f-grp"><label className="f-lbl">{t.jobTitle}</label><input className="f-inp" placeholder={lang==="ru"?"Укладка плитки в ванной":lang==="kz"?"Ваннаға кафель төсеу":"Bathroom tiling"}/></div>
              <div className="f-grp"><label className="f-lbl">{t.jobDesc}</label><textarea className="f-inp f-ta" placeholder={lang==="ru"?"Подробное описание...":lang==="kz"?"Толық сипаттама...":"Detailed description..."}/></div>
              <div className="f-grp"><label className="f-lbl">{t.budget}</label><input className="f-inp" placeholder="50 000 тг"/></div>
              <div className="f-grp">
                <label className="f-lbl">{t.catsLabel}</label>
                <div className="cat-grid">
                  {CATEGORIES.slice(0,6).map(cat=>(
                    <div key={cat.id} className={`cat-chip ${selCats.includes(cat.id)?"sel":""}`} onClick={()=>setSelCats(p=>p.includes(cat.id)?p.filter(c=>c!==cat.id):[...p,cat.id])}>
                      <span className="cc-icon">{cat.icon}</span><span className="cc-name">{getCatName(cat)}</span>
                    </div>
                  ))}
                </div>
              </div>
              <button className="act-btn" onClick={()=>setShowPost(false)}>{t.publish}</button>
            </div>
          </div>
        )}

        {/* ── CITY MODAL ── */}
        {showCity && (
          <div className="overlay" onClick={()=>setShowCity(false)}>
            <div className="sheet" onClick={e=>e.stopPropagation()}>
              <div className="sheet-ttl">{t.selectCity}</div>
              {cities.map(c=>(
                <div key={c} className={`city-opt ${city===c?"sel":""}`} onClick={()=>{setCity(c);setShowCity(false);}}>
                  <span style={{fontSize:20}}>📍</span>
                  <span className="city-opt-name">{c}</span>
                  {city===c && <Ic path={icons.check} size={18} color="var(--blue)" style={{marginLeft:"auto"}}/>}
                </div>
              ))}
            </div>
          </div>
        )}

      </div>
    </>
  );
}
